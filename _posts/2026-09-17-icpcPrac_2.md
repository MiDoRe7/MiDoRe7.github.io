## 1. 함수 준비
기하 관련 문제를 풀기 위해 몇 가지 함수들을 미리 준비하기로 했다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const ll MOD = 1e9 + 7;

// 빠른 거듭제곱: base^exp % MOD.  O(log exp)
ll power(ll base, ll exp, ll mod = MOD) {
    base %= mod;
    ll res = 1;
    while (exp > 0) {
        if (exp & 1) res = res * base % mod;
        base = base * base % mod;
        exp >>= 1;
    }
    return res;
}

// 모듈러 역원 (MOD가 소수일 때, 페르마 소정리).  a^(MOD-2) ≡ a^(-1)
ll inv(ll a, ll mod = MOD) {
    return power(a, mod - 2, mod);
}

// 확장 유클리드: ax + by = gcd(a,b) 의 (g, x, y) 반환. MOD가 소수가 아닐 때 역원용
ll extgcd(ll a, ll b, ll& x, ll& y) {
    if (b == 0) { x = 1; y = 0; return a; }
    ll x1, y1;
    ll g = extgcd(b, a % b, x1, y1);
    x = y1;
    y = x1 - (a / b) * y1;
    return g;
}

// --- 조합론: 팩토리얼 전처리 후 nCr을 O(1)로 ---
const int MAXN = 1000005;
ll fact[MAXN], invfact[MAXN];

void initComb() {
    fact[0] = 1;
    for (int i = 1; i < MAXN; i++) fact[i] = fact[i-1] * i % MOD;
    invfact[MAXN-1] = inv(fact[MAXN-1]);
    for (int i = MAXN-2; i >= 0; i--) invfact[i] = invfact[i+1] * (i+1) % MOD;
}

ll nCr(int n, int r) {
    if (r < 0 || r > n) return 0;
    return fact[n] * invfact[r] % MOD * invfact[n-r] % MOD;
}
```

## 2. Fermat's Little Theorem
조건 : 소수 $p$, $gcd(a, p) = 1$인 정수 $a$
$$a^{p-1} \equiv 1 \pmod p$$

Binary Exponentiation을 사용해 $O(log P)$ 안에 모듈러 역원을 구할 수 있는 방법이 있다.
```cpp
#include <iostream>
using namespace std;
const long long MOD = 1000000007; // 소수 P

// 밑(base)의 exp 제곱을 mod로 나눈 나머지를 O(log exp)에 계산하는 함수
long long power(long long base, long long exp, long long mod) {
    long long result = 1;
    base %= mod;

    while (exp > 0) {
        // 현재 비트가 1이면(지수가 홀수이면) 결과에 곱해줌
        if (exp % 2 == 1) {
            result = (result * base) % mod;
        }
        // 밑을 제곱하고 지수를 절반으로 줄임
        base = (base * base) % mod;
        exp /= 2;
    }

    return result;
}

// 페르마의 소정리를 이용해 n의 모듈러 역원을 구하는 함수
// n과 p는 서로소여야 하며, p는 소수여야 함
long long modInverse(long long n, long long p) {
    return power(n, p - 2, p);
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    long long A = 12;
    long long B = 3;

    // (A / B) mod MOD 계산하기
    // 나눗셈을 B의 모듈러 역원을 곱하는 형태로 변환
    long long invB = modInverse(B, MOD);
    long long answer = (A % MOD * invB) % MOD;

    cout << "B의 모듈러 역수: " << invB << "\n";
    cout << "(A / B) mod MOD 결과: " << answer << "\n";

    // 역원이 맞는지 역검증: (B * invB) % MOD는 반드시 1이어야 함
    cout << "검증 (B * invB) % MOD: " << (B * invB) % MOD << "\n";

    return 0;
}
```

## 3. Extended Euclidean Algorithm
#### Bézout's Identity
두 정수 $A, M$에 대해 다음 식을 만족하는 정수해 $x, y$가 항상 존재한다.
$$A \cdot x + M \cdot y = \gcd(A, M)$$

이 때 $gcd(A, M) = 1$이고, 양 변에 모듈러 $M$을 취하면
$$A \cdot x \equiv 1 \pmod M$$

```cpp
#include <iostream>

using namespace std;

// 확장 유클리드 호제법: a * x + b * y = gcd(a, b)를 만족하는 gcd, x, y를 계산
long long extGCD(long long a, long long b, long long &x, long long &y) {
    if (b == 0) {
        x = 1;
        y = 0;
        return a;
    }
    long long x1, y1;
    long long gcd = extGCD(b, a % b, x1, y1);
    
    // 이전 단계의 해를 이용해 현재 단계의 x, y 복원
    x = y1;
    y = x1 - (a / b) * y1;
    
    return gcd;
}

// 모듈러 M이 소수가 아닐 때도 동작하는 모듈러 역원 함수
// 역원이 존재하지 않으면 -1을 반환
long long modInverse(long long a, long long m) {
    long long x, y;
    long long gcd = extGCD(a, m, x, y);

    // a와 m이 서로소가 아니면 모듈러 역원이 존재하지 않음
    if (gcd != 1) {
        return -1;
    }

    // x가 음수일 수 있으므로 m 범위 내의 양수로 정규화
    return (x % m + m) % m;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    long long A = 3;
    long long M = 10; // 10은 소수가 아닌 합성수

    long long inv = modInverse(A, M);

    if (inv == -1) {
        cout << A << "의 mod " << M << "에 대한 역원이 존재하지 않습니다.\n";
    } else {
        cout << A << "의 mod " << M << " 역원: " << inv << "\n";
        cout << "검증 (" << A << " * " << inv << ") % " << M << " = " 
             << (A * inv) % M << "\n";
    }

    return 0;
}
```

확장 유클리드 호제법이 반환하는 $x$는 음수일 수 있기 때문에 $(x \pmod m + m) \pmod m$ 연산을 통해 양수화할 필요가 있다.
시간 복잡도는 $$O(log(min(A, M))$$이다.

## 4. Chinese Remainder Theorem
어떤 수 x를 서로 다른 수들로 나누었을 때의 나머지를
$x \equiv a_1 \pmod{m_1}$
$x \equiv a_2 \pmod{m_2}$
$\dots$
$x \equiv a_k \pmod{m_k}$
라고 하자.

모든 모듈러 값들이 서로의 쌍마다 서로소($\gcd(m_i, m_j) = 1$)라면, $M = m_1 \times m_2 \times \dots \times m_k$에 대해 $0 \le x < M$ 범위 내에서 유일한 정수해 $x$가 반드시 존재한다.

$M = m_1 \times m_2 \times \dots \times m_k$을 계산하고,
$M_i = M / m_i$를 구한다.

확장 유클리드 호제법을 사용해 $M_i \times y_i \equiv 1 \pmod{m_i}$로 역원을 구한다.
이후 각 조건의 나머지, 독립 항인 $M_i$, 역원 보정값 $y_i$를 곱해 모두 더한 뒤 $M$에 대한 나머지를 취한다.
$x = \sum_{i=1}^k (a_i \times M_i \times y_i) \pmod M$

$k$번째 식에 $x$를 대입했을 때 다른 항들은 $m_k$의 배수이기 때문에 사라지고, $a_k$만 남아 모든 합동식을 만족하게 된다.

```cpp
#include <iostream>
#include <vector>

using namespace std;

// 확장 유클리드 호제법
long long extGCD(long long a, long long b, long long &x, long long &y) {
    if (b == 0) {
        x = 1;
        y = 0;
        return a;
    }
    long long x1, y1;
    long long gcd = extGCD(b, a % b, x1, y1);
    x = y1;
    y = x1 - (a / b) * y1;
    return gcd;
}

// 모듈러 역원 계산기
long long modInverse(long long a, long long m) {
    long long x, y;
    long long gcd = extGCD(a, m, x, y);
    if (gcd != 1) return -1; // 역원이 없는 경우
    return (x % m + m) % m;
}

// 중국인의 나머지 정리 함수
// a: 나머지 배열, m: 법(mod) 배열
long long chineseRemainderTheorem(const vector<long long>& a, const vector<long long>& m) {
    int k = a.size();
    long long M = 1;

    // 1. 전체 M 계산
    for (int i = 0; i < k; i++) {
        M *= m[i];
    }

    long long result = 0;

    // 2. 각 항을 조립하여 누적
    for (int i = 0; i < k; i++) {
        long long Mi = M / m[i];
        long long yi = modInverse(Mi, m[i]);

        // (a[i] * Mi * yi) mod M 계산 시 오버플로우 방지
        long long term = (a[i] % M * (Mi % M)) % M;
        term = (term * (yi % M)) % M;

        result = (result + term) % M;
    }

    return (result + M) % M;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    
    vector<long long> a = {2, 3, 2};
    vector<long long> m = {3, 5, 7};

    long long answer = chineseRemainderTheorem(a, m);

    cout << "동시에 만족하는 최소 정수 x: " << answer << "\n";

    return 0;
}
```
CRT를 사용하면 여러 모듈러 조건을 동시에 만족하는 수를 찾아낼 수 있다.
