## 1. 함수 준비
기하 관련 문제를 풀기 위해 몇 가지 함수들을 미리 준비하기로 했다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

struct P {
    ll x, y;
    P operator-(const P& o) const { return { x - o.x, y - o.y }; }
    ll cross(const P& o) const { return x * o.y - y * o.x; }   // 외적
    ll dot(const P& o) const { return x * o.x + y * o.y; }     // 내적
};

// 세 점 a->b->c 의 방향. >0 반시계(왼쪽), <0 시계(오른쪽), 0 일직선
ll ccw(const P& a, const P& b, const P& c) {
    return (b - a).cross(c - a);
}
int sgn(ll v) { return (v > 0) - (v < 0); }

// 점 p 가 선분 ab 위에 있는가 (일직선 && 범위 안)
bool onSegment(const P& a, const P& b, const P& p) {
    if (ccw(a, b, p) != 0) return false;
    return min(a.x, b.x) <= p.x && p.x <= max(a.x, b.x) &&
        min(a.y, b.y) <= p.y && p.y <= max(a.y, b.y);
}

// 두 선분 ab, cd 가 교차하는가 (끝점 접촉 포함)
bool segIntersect(const P& a, const P& b, const P& c, const P& d) {
    int d1 = sgn(ccw(a, b, c)), d2 = sgn(ccw(a, b, d));
    int d3 = sgn(ccw(c, d, a)), d4 = sgn(ccw(c, d, b));
    if (d1 * d2 < 0 && d3 * d4 < 0) return true;   // 정상 교차
    if (d1 == 0 && onSegment(a, b, c)) return true;
    if (d2 == 0 && onSegment(a, b, d)) return true;
    if (d3 == 0 && onSegment(c, d, a)) return true;
    if (d4 == 0 && onSegment(c, d, b)) return true;
    return false;
}

// 다각형 넓이의 2배 (신발끈). 실제 넓이는 이 값 / 2.0. 정수로 보관해 오차 없음
ll area2(const vector<P>& poly) {
    ll s = 0;
    int n = poly.size();
    for (int i = 0; i < n; i++)
        s += poly[i].cross(poly[(i + 1) % n]);
    return llabs(s);
}

// 볼록껍질 (Andrew's monotone chain). 반시계 방향 정점 목록 반환
vector<P> convexHull(vector<P> pts) {
    sort(pts.begin(), pts.end(), [](const P& a, const P& b) {
        return a.x != b.x ? a.x < b.x : a.y < b.y;
        });
    pts.erase(unique(pts.begin(), pts.end(), [](const P& a, const P& b) {
        return a.x == b.x && a.y == b.y;
        }), pts.end());
    int n = pts.size();
    if (n < 3) return pts;
    vector<P> h(2 * n);
    int k = 0;
    for (int i = 0; i < n; i++) {          // 아래쪽 체인
        while (k >= 2 && ccw(h[k - 2], h[k - 1], pts[i]) <= 0) k--;
        h[k++] = pts[i];
    }
    for (int i = n - 2, lo = k + 1; i >= 0; i--) {  // 위쪽 체인
        while (k >= lo && ccw(h[k - 2], h[k - 1], pts[i]) <= 0) k--;
        h[k++] = pts[i];
    }
    h.resize(k - 1);
    return h;
}
```

간단한 문제들을 풀며 함수의 검증은 완료되었지만, Convex Hull 문제의 조건이 변 위에 있는 정점을 포함하는 걸 요구할 경우 ccw(h[k-2], h[k-1], pts[i])<0)처럼 <= 기호를 <로 수정할 필요가 있다.

## 2. Point In Polygon
Ray Casting Algorithm을 사용하여 구현: 특정 점 p에서 임의 방향으로 Ray를 쏘았을 때 다각형의 변들과 만나는 교점의 개수를 세어 결과값이 홀수 개면 다각형의 내부(1), 짝수 개면 외부(-1)로 판정
```cpp
int pointInPolygon(const vector<P>& poly, const P& p) {
	int n = poly.size();
	bool inside = false;
	for (int i = 0; i < n; i++) {
		const P& a = poly[i];
		const P& b = poly[(i + 1) % n];
		if (onSegment(a, b, p)) return 0;
        if (a.y <= p.y) {
            if (b.y > p.y && ccw(a, b, p) > 0) inside = !inside;  // 위로 향하는 변, p가 왼쪽
        }
        else {
            if (b.y <= p.y && ccw(a, b, p) < 0) inside = !inside;  // 아래로 향하는 변, p가 오른쪽
        }
	}
	return inside ? 1 : -1;
}
```
정확히 경계에 걸친 경우를 먼저 onSegment로 배제하고, Ray Casting을 사용하여 Inside/Outside를 판정했다.
